#include<iostream>
#include<vector>
#include<unordered_map>
#include<unordered_set>
#include<climits>
#include<cstdlib>
#include<random>
#include<fstream>
using namespace std;

class customHash{
    public:
        size_t operator()(const pair<int,int> a) const{
            size_t first=a.first;
            size_t second=a.second;
            
            return first <<(first^second);
        }
};


class Graph{
    
    int n;
    unordered_map<int,unordered_set<int>> adjlist;
    vector<pair<int,int>> edges;
    
    public:
        Graph(int n){
            this->n=n;
        }
        
        void insertEdge(int u,int v){
            if(u>this->n || u<1 || v>this->n || v<1){
                cout<<"edges must be between 1" << n<<endl;
                return;
            }
            adjlist[u].insert(v);
            adjlist[v].insert(u);
            
           edges.push_back({u,v});
           edges.push_back({v,u});
            
        }
        
        void showAdjList(){
             
             for(auto it=adjlist.begin();it!=adjlist.end();it++){
                 cout<<it->first<<"-> ";
                 for(auto &i:it->second){
                     cout<<i<<" ";
                 }
                 cout<<endl;
             }
        }
        
       bool checkIsVertexCover(vector<int> &sub){
          
            
            bool valid=true;
            for( auto [u,v]:this->edges){
                
                bool found=false;
                for(int i=0;i<sub.size();i++){
                    if(sub[i]==u || sub[i]==v){
                        found=true;
                    }
                }
                
                if(!found){
                    valid=false;
                }
                
                
            }
            
            
            return valid;
          
            
            
            
        }
        void createVertexSet(int V,vector<int> &temp,vector<int> &ans,int i=1){
            if(i>V){
                
                auto validVertex=checkIsVertexCover(temp);
                
                if(validVertex){
                    
                    if( ans.size()==0 || temp.size()<ans.size()){
                        ans=temp;
                    }
                }
                return;
            }
            
            
            
            temp.push_back(i);
            createVertexSet(V,temp,ans,i+1);
            temp.pop_back();
            
            createVertexSet(V,temp,ans,i+1);

        }
        void dfs(int src, vector<bool> &visited){
            
            visited[src]=true;
            
            for(auto &j:adjlist[src]){
                dfs(j,visited);
            }
        }
        bool isConnected(){
            vector<bool> visited(n+1,false);
            int count=0;
            for(int i=1;i<=n;i++){
                if(!visited[i]){
                    count++;
                    
                    dfs(i,visited);
                }
            }
            
            
            return count==1;
        }
        
        void findVertexCover(){
            fstream fileTwo("output.txt",std::ios::app);
            vector<int> temp;
            
            vector<int> vertexCover;
            
            createVertexSet(this->n,temp,vertexCover);

            fileTwo<<"[";
            for(int i=0;i<vertexCover.size();i++){
                fileTwo<<vertexCover[i];
                if(i!=vertexCover.size()-1){
                    fileTwo<<",";
                }
            }
            
            fileTwo<<"]"<<endl;
            
            fileTwo<<"Vertex Cover is of size "<<vertexCover.size()<<endl<<endl;
        
        }

};
int main(){
    
    int n=10;
    
    fstream fileOne("input.txt",std::ios::app);
    Graph *g=NULL;
   
    srand(time(0));
    vector<int> noOfEdges={10,15,20,25,30,35,40,45};
    int E=noOfEdges[rand()%(noOfEdges.size())];
    cout<<"total edges "<< E<<endl;
    
    while(true){
        cout<<"started"<<endl;
        unordered_map<pair<int,int>,bool,customHash> isValidEdge;
        vector<pair<int,int>> edges;
        g=new Graph(n);
        int count=0;
    
        for(int i=0;;i++){
            int u=(rand() %10)+1;
            int v= (rand()%10)+1;
            if(u==v) continue;
            if(count==E) break;
            if(isValidEdge[{u,v}] || isValidEdge[{v,u}]){
                continue;
            }
            edges.push_back({u,v});
            g->insertEdge(u,v);
            count++;
            isValidEdge[{u,v}]=true;
            isValidEdge[{v,u}]=true;
   
        }
        
        bool isConnect=g->isConnected();
        
        if(!isConnect) {
            delete g;
            g=NULL;
            continue;
            
        }
        //  fileOne<<"Edges=[";
        //  fileOne<< "{" << u<< ","<<v<<"}";
        //  if(count!=E) fileOne<<" , ";
        // fileOne<<"]"<<endl<<endl;
        
        cout<<"completed"<<endl;
        
        
        
        break;
    }
    
    
    g->showAdjList();
    g->findVertexCover();
  
    
    
    return 0;
}
