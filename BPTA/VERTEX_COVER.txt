class customHash{
    public:
        size_t operator()( const pair<int,int> a) const{
            size_t first=a.first;
            size_t second=a.second;
            
            return first<< (first^second);
        }
};


class Graph{
    
    int n;
    unordered_map<int,unordered_set<int>> adjlist;
    vector<pair<int,int>> edges;
    
    public:
        Graph(int n){
            this->n=n;
        }
        
        void insertEdge(int u,int v){
            if(u>this->n || u<1 || v>this->n || v<1){
                cout<<"edges must be between 1" << n<<endl;
            }
            adjlist[u].insert(v);
            adjlist[v].insert(u);
            
           edges.push_back({u,v});
           edges.push_back({v,u});
            
        }
        
        void showAdjList(){
             
             for(auto it=adjlist.begin();it!=adjlist.end();it++){
                 cout<<it->first<<"-> ";
                 for(auto &i:it->second){
                     cout<<i<<" ";
                 }
                 cout<<endl;
             }
        }
        
       bool checkIsVertexCover(vector<int> &sub){
          
            
            bool valid=true;
            for( auto [u,v]:this->edges){
                
                bool found=false;
                for(int i=0;i<sub.size();i++){
                    if(sub[i]==u || sub[i]==v){
                        found=true;
                    }
                }
                
                if(!found){
                    valid=false;
                }
                
                
            }
            
            
            return valid;
          
            
            
            
        }
        void createVertexSet(int V,vector<int> &temp,vector<int> &ans,int i=1){
            if(i>V){
                
                auto validVertex=checkIsVertexCover(temp);
                
                if(validVertex){
                    
                    if( ans.size()==0 || temp.size()<=ans.size()){
                        ans=temp;
                    }
                }
                return;
            }
            
            
            
            temp.push_back(i);
            createVertexSet(V,temp,ans,i+1);
            temp.pop_back();
            
            createVertexSet(V,temp,ans,i+1);

        }
        
        
        void findVertexCover(){
            vector<int> temp;
            
            vector<int> vertexCover;
            
            createVertexSet(this->n,temp,vertexCover);
            
            
            
            
            cout<<"Vertex Cover is of size "<<vertexCover.size()<<endl;
            
            
            for(auto &j:vertexCover){
                cout<<j<<" ";
            }
            
            
            
            
        }

};
int main(){
    
    int n=10;
    Graph g(n);
   
    srand(time(0));
    
    unordered_map<pair<int,int>,bool,customHash> isValidEdge;
    // vector<pair<int,int>> edges={ {1,2},{1,4} ,{2,3} ,{3,5},{3,8}, {2,4},{4,7},{7,8} ,{8,5},{1,6},{6,7} };
    // for(auto &[u,v]:edges){
    //     g.insertEdge(u,v);
    // }
    // g.showAdjList();
    // g.findVertexCover();
    int m=rand()%(36)+10;
    
    for(int i=0;i<m;i++){
        int u=(rand() %9)+1;
        int v= (rand()%9)+1;
        
        if((isValidEdge.size()/2)==m) break;
        
        
        if(isValidEdge[{u,v}] || isValidEdge[{v,u}]){
            continue;
        }
        g.insertEdge(u,v);
        isValidEdge[{u,v}]=true;
        isValidEdge[{v,u}]=true;
        
    }
    
    g.showAdjList();
    g.findVertexCover();
    return 0;
}
