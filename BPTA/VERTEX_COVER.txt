#include <iostream>
#include <vector>
#include <unordered_map>
#include <unordered_set>
#include <cstdlib>
#include <ctime>
#include <fstream>
#include <utility>

using namespace std;

class CustomHash {
public:
    size_t operator()(const pair<int, int>& p) const {
        return static_cast<size_t>(p.first) ^
               (static_cast<size_t>(p.second) << 1);
    }
};

class Graph {

    int n;
    unordered_map<int, unordered_set<int>> adjlist;
    vector<pair<int, int>> edges;

public:

    Graph(int n) {
        this->n = n;
    }

    void insertEdge(int u, int v) {

        if (u < 1 || u > n || v < 1 || v > n) {
            cout << "Edges must be between 1 and " << n << endl;
            return;
        }

        if (u == v) {
            return;
        }

        // Avoid duplicate edges
        if (adjlist[u].count(v)) {
            return;
        }

        adjlist[u].insert(v);
        adjlist[v].insert(u);

        // Store an undirected edge only once
        edges.push_back({u, v});
    }

    void showAdjList() {

        for (auto& [vertex, neighbours] : adjlist) {

            cout << vertex << " -> ";

            for (int neighbour : neighbours) {
                cout << neighbour << " ";
            }

            cout << endl;
        }
    }

    bool checkIsVertexCover(const vector<int>& sub) {

        vector<bool> selected(n + 1, false);

        for (int vertex : sub) {
            selected[vertex] = true;
        }

        for (auto [u, v] : edges) {

            // Neither endpoint is selected
            if (!selected[u] && !selected[v]) {
                return false;
            }
        }

        return true;
    }

    void createVertexSet(
        int V,
        vector<int>& temp,
        vector<int>& ans,
        int i = 1
    ) {

        // All vertices have been considered
        if (i > V) {

            if (checkIsVertexCover(temp)) {

                if (ans.empty() || temp.size() < ans.size()) {
                    ans = temp;
                }
            }

            return;
        }

        // Include vertex i
        temp.push_back(i);

        createVertexSet(V, temp, ans, i + 1);

        temp.pop_back();

        // Don't include vertex i
        createVertexSet(V, temp, ans, i + 1);
    }

    void dfs(int src, vector<bool>& visited) {

        visited[src] = true;

        for (int neighbour : adjlist[src]) {

            if (!visited[neighbour]) {
                dfs(neighbour, visited);
            }
        }
    }

    bool isConnected() {

        vector<bool> visited(n + 1, false);

        int components = 0;

        for (int i = 1; i <= n; i++) {

            if (!visited[i]) {

                components++;

                dfs(i, visited);
            }
        }

        return components == 1;
    }

    void findVertexCover() {

        ofstream file("output.txt", ios::app);

        vector<int> temp;
        vector<int> vertexCover;

        createVertexSet(n, temp, vertexCover);

        file << "[";

        for (size_t i = 0; i < vertexCover.size(); i++) {

            file << vertexCover[i];

            if (i != vertexCover.size() - 1) {
                file << ",";
            }
        }

        file << "]\n";

        file << "Vertex Cover is of size "
             << vertexCover.size()
             << "\n\n";
    }
};


int main() {

    int n = 10;

    Graph* g = nullptr;

    srand(static_cast<unsigned>(time(nullptr)));

    vector<int> possibleEdges = {
        10, 15, 20, 25, 30, 35, 40, 45
    };

    int E = possibleEdges[rand() % possibleEdges.size()];

    cout << "Total edges: " << E << endl;

    while (true) {

        cout << "Started" << endl;

        unordered_map<pair<int, int>, bool, CustomHash> isValidEdge;

        g = new Graph(n);

        int count = 0;

        while (count < E) {

            int u = (rand() % n) + 1;
            int v = (rand() % n) + 1;

            // No self-loop
            if (u == v) {
                continue;
            }

            // No duplicate edge
            if (isValidEdge[{u, v}] ||
                isValidEdge[{v, u}]) {
                continue;
            }

            g->insertEdge(u, v);

            isValidEdge[{u, v}] = true;
            isValidEdge[{v, u}] = true;

            count++;
        }

        // Keep only connected graphs
        if (!g->isConnected()) {

            delete g;
            g = nullptr;

            continue;
        }

        cout << "Completed" << endl;

        break;
    }

    g->showAdjList();

    g->findVertexCover();

    delete g;
    g = nullptr;

    return 0;
}
